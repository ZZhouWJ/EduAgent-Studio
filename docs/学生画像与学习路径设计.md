# 学生画像与学习路径设计

## 概述

本文档描述 EduAgent Studio 系统中学生画像数据模型与学习路径规划的核心设计。学生画像是个性化学习的基石，通过采集学生的基础信息、学习行为和表现数据，构建多维度画像，从而为后续的资源推荐和路径规划提供决策依据。

## 3. 掌握度计算算法

### 3.1 知识点级掌握度算法

```python
def calculate_kp_mastery(quiz_score: float, self_mastery: float) -> float:
    """
    计算单个知识点的掌握度
    
    参数:
        quiz_score: 测验得分 (0-100)
        self_mastery: 学生自评掌握度 (0-100)
    
    返回:
        mastery_level: 掌握度 (0.0 - 1.0)
    
    公式: quiz_score * 0.7 + self_mastery * 0.3
    """
    quiz_weight = 0.7
    self_weight = 0.3
    
    # 归一化到 0-1 范围
    mastery = (quiz_score * quiz_weight + self_mastery * self_weight) / 100.0
    
    # 边界裁剪
    return max(0.0, min(1.0, mastery))
```

**算法说明：**

- **权重设计理由**：测验成绩具有客观性权重 70%，自评具有主观性权重 30%
- **动态调整**：随着测验次数增加，quiz_score 权重可适当提高
- **防作弊机制**：连续满分但自评低时触发人工复核

### 3.2 综合掌握度算法

```python
def calculate_overall_mastery(student_id: str, db: Session) -> float:
    """
    计算学生的综合掌握度
    
    返回:
        overall: 综合掌握度 (0.0 - 1.0)，即所有知识点掌握度的平均值
    """
    mastery_records = db.query(StudentKnowledgeMastery).filter(
        StudentKnowledgeMastery.student_id == student_id
    ).all()
    
    if not mastery_records:
        return 0.0
    
    total_mastery = sum(r.mastery_level for r in mastery_records)
    return total_mastery / len(mastery_records)
```

### 3.3 薄弱点判断

```python
WEAK_POINT_THRESHOLD = 0.5  # 薄弱点阈值

def update_weak_points(student_id: str, db: Session) -> List[str]:
    """
    更新学生的薄弱知识点列表
    
    返回:
        weak_kp_ids: 薄弱知识点ID列表
    """
    records = db.query(StudentKnowledgeMastery).filter(
        StudentKnowledgeMastery.student_id == student_id
    ).all()
    
    weak_points = []
    for record in records:
        # 判断是否需要更新薄弱点标记
        is_weak = record.mastery_level < WEAK_POINT_THRESHOLD
        if record.is_weak_point != is_weak:
            record.is_weak_point = is_weak
            db.add(record)
        
        if is_weak:
            weak_points.append(str(record.knowledge_point_id))
    
    db.commit()
    return weak_points
```

### 4.2 优先级排序算法

```python
from typing import List, Dict, Tuple
from collections import defaultdict, deque

class LearningPathPlanner:
    """学习路径规划器"""
    
    def __init__(self, db: Session, student_id: str):
        self.db = db
        self.student_id = student_id
        self.weak_threshold = 0.5
    
    def generate_learning_path(self) -> List[Dict]:
        """
        生成个性化学习路径
        
        排序策略:
        1. 薄弱点优先 (weak_points_first)
        2. 先修依赖拓扑排序 (topological_sort)
        3. 综合权重排序 (comprehensive_scoring)
        """
        # 1. 获取所有知识点的掌握度
        mastery_map = self._get_mastery_map()
        
        # 2. 获取知识点依赖关系
        dependency_graph = self._build_dependency_graph()
        
        # 3. 计算每个知识点的优先级得分
        priority_scores = self._calculate_priority_scores(mastery_map, dependency_graph)
        
        # 4. 拓扑排序 + 优先级融合
        sorted_kps = self._topological_priority_sort(dependency_graph, priority_scores)
        
        # 5. 为每个知识点匹配资源
        path = self._match_resources(sorted_kps, mastery_map)
        
        return path
    
    def _calculate_priority_scores(
        self, 
        mastery_map: Dict[str, float],
        dependency_graph: Dict[str, List[str]]
    ) -> Dict[str, float]:
        """
        计算每个知识点的优先级得分
        
        公式: score = weak_factor * 0.5 + prerequisite_factor * 0.3 + difficulty_factor * 0.2
        """
        scores = {}
        
        for kp_id, mastery in mastery_map.items():
            # 薄弱因子：掌握度越低，优先级的潜力越高
            weak_factor = max(0, self.weak_threshold - mastery) / self.weak_threshold
            
            # 前置依赖满足因子：前置知识已掌握则加分
            prereq_factor = self._calculate_prereq_satisfaction(kp_id, dependency_graph, mastery_map)
            
            # 难度因子：适中难度优先
            difficulty = self._get_kp_difficulty(kp_id)
            difficulty_factor = 1.0 - abs(difficulty - 3) / 5  # 3 为最佳难度
            
            # 综合得分
            scores[kp_id] = weak_factor * 0.5 + prereq_factor * 0.3 + difficulty_factor * 0.2
        
        return scores
    
    def _topological_priority_sort(
        self,
        graph: Dict[str, List[str]],
        scores: Dict[str, float]
    ) -> List[str]:
        """
        拓扑排序与优先级融合
        
        使用改进的 Kahn 算法，在同层级内按优先级排序
        """
        # 计算入度
        in_degree = defaultdict(int)
        all_nodes = set(graph.keys())
        for node, parents in graph.items():
            for parent in parents:
                all_nodes.add(parent)
        
        for node, parents in graph.items():
            for parent in parents:
                in_degree[node] += 1
        
        # 初始化队列：将入度为0的节点按优先级排序
        queue = [node for node in all_nodes if in_degree[node] == 0]
        queue.sort(key=lambda x: scores.get(x, 0), reverse=True)
        queue = deque(queue)
        
        sorted_order = []
        
        while queue:
            # 按优先级从队列中取出
            current = queue.popleft()
            sorted_order.append(current)
            
            # 更新相关节点入度
            for node, parents in graph.items():
                if current in parents:
                    in_degree[node] -= 1
                    if in_degree[node] == 0:
                        # 按优先级插入
                        self._insert_by_priority(queue, node, scores)
        
        return sorted_order
    
    def _insert_by_priority(self, queue: deque, node: str, scores: Dict[str, float]):
        """按优先级插入队列（保持队列有序）"""
        inserted = False
        for i, q_node in enumerate(queue):
            if scores.get(node, 0) > scores.get(q_node, 0):
                queue.insert(i, node)
                inserted = True
                break
        if not inserted:
            queue.append(node)
```

### 4.3 资源类型匹配算法

```python
from enum import Enum

class ResourceType(Enum):
    LECTURE_NOTES = "lecture_notes"    # 讲义
    VIDEO = "video"                    # 视频
    CASE_STUDY = "case_study"          # 案例
    HANDS_ON = "hands_on"             # 实操

def match_resource_type(
    mastery_level: float,
    cognitive_style: str,
    resource_preferences: Dict[str, float]
) -> List[ResourceType]:
    """
    根据掌握度和个人偏好匹配资源类型
    
    策略:
    - 掌握度 < 0.3: 推荐讲义 + 视频（打基础）
    - 掌握度 0.3-0.6: 推荐视频 + 案例（强化理解）
    - 掌握度 > 0.6: 推荐案例 + 实操（提升应用）
    
    同时考虑认知风格和资源偏好
    """
    base_types = []
    
    if mastery_level < 0.3:
        base_types = [ResourceType.LECTURE_NOTES, ResourceType.VIDEO]
    elif mastery_level < 0.6:
        base_types = [ResourceType.VIDEO, ResourceType.CASE_STUDY]
    else:
        base_types = [ResourceType.CASE_STUDY, ResourceType.HANDS_ON]
    
    # 根据认知风格调整
    style_adjustments = {
        "visual": [ResourceType.VIDEO, ResourceType.LECTURE_NOTES],
        "auditory": [ResourceType.VIDEO],
        "kinesthetic": [ResourceType.HANDS_ON, ResourceType.CASE_STUDY]
    }
    
    # 合并调整
    preferred = style_adjustments.get(cognitive_style, [])
    for res_type in preferred:
        if res_type not in base_types:
            base_types.append(res_type)
    
    # 根据资源偏好权重二次排序
    type_scores = {}
    for res_type in base_types:
        pref_key = res_type.value
        type_scores[res_type] = resource_preferences.get(pref_key, 0.5)
    
    # 按偏好得分降序排列
    base_types.sort(key=lambda x: type_scores[x], reverse=True)
    
    return base_types
```

| 0.3 - 0.6 | 成长期 | 视频 + 案例 | 深化理解，学会应用 |
| 0.6 - 1.0 | 精通期 | 案例 + 实操 | 巩固提升，综合运用 |

```tsx
// RadarChart.tsx
interface RadarChartProps {
  data: {
    label: string;
    value: number;  // 0-10
  }[];
}

const RadarChart: React.FC<RadarChartProps> = ({ data }) => {
  const centerX = 150;
  const centerY = 150;
  const maxRadius = 100;
  
  const angleStep = (2 * Math.PI) / data.length;
  
  // 计算多边形顶点
  const getPoint = (index: number, value: number) => {
    const angle = index * angleStep - Math.PI / 2;
    const radius = (value / 10) * maxRadius;
    return {
      x: centerX + radius * Math.cos(angle),
      y: centerY + radius * Math.sin(angle)
    };
  };
  
  // 数据多边形点
  const dataPoints = data.map((d, i) => getPoint(i, d.value));
  const dataPath = dataPoints.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
  
  return (
    <svg width="300" height="300" viewBox="0 0 300 300">
      {/* 背景网格 */}
      {[2, 4, 6, 8, 10].map((level) => (
        <polygon
          key={level}
          points={data.map((_, i) => {
            const p = getPoint(i, level);
            return `${p.x},${p.y}`;
          }).join(' ')}
          fill="none"
          stroke="#e5e7eb"
          strokeWidth="1"
        />
      ))}
      
      {/* 轴线 */}
      {data.map((_, i) => {
        const p = getPoint(i, 10);

            key={i}
            x1={centerX}
            y1={centerY}
            x2={p.x}
            y2={p.y}
            stroke="#e5e7eb"
          />

      })}
      
      {/* 数据多边形 */}

        stroke="#3b82f6"
        strokeWidth="2"
      />
```

```
POST /api/v1/mastery/update

Request Body:
{
  "student_id": "2026001",
  "knowledge_point_id": "kp001",
  "quiz_score": 78.5,
  "self_mastery": 70.0
}

Response 200:
{
  "success": true,
  "knowledge_point_id": "kp001",
  "new_mastery_level": 0.759,  // 78.5*0.7 + 70*0.3 = 75.95 → 0.7595
  "is_weak_point": false,      // 0.759 > 0.5, 不再是薄弱点
  "overall_mastery_updated": true,
  "new_overall_mastery": 73.2
}
```
